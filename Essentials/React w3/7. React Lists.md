In React, you will render lists with some type of loop.
The JavaScript `map()` array method is generally the preferred method.
```jsx
import "./App.scss";
  
function car_List(){
  const cars=["Mustang","BMW","Ferrari"];
  return(
    <div>
      <h1>My cars:</h1>
      <ul>
        {cars.map((car)=><li>Car's brand: {car}</li>)}
        //or
        {cars.map((car,index)=><li key={index}>car:{car}</li>)}
      </ul>
    </div>
  )
}
  
export default function App(){
  return(
    <div className="App">
      {car_List()}
    </div>
  );
}
```
