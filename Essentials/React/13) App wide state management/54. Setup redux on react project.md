##### Install redux in react
We can work with the redux-toolkit but we need to learn the basics first. So we install the react-redux package first.

First, install `redux`, and then the `react-redux` package.
```bash
npm install redux
npm install react-redux

//or

npm i react-redux redux
```
`react-redux` helps us to use redux easier in our project.
##### Get started
Create a folder in your `src` folder called `redux`(or whatever you see fit).
Then a file in it.(You can name it store).
Then a folder for our reducers. Here we want to complete our counter example. So create a file in it called `counterReducer`.

For start, let's create our reducer:
```js
const initialVal={counter:0};

const INCREASE="increase";
const DECREASE="decrease";
const RESET="reset";

export const counterReducer=(state=initialVal,action)=>{
    switch(action.type){
        case INCREASE:
            return{
                counter:state.counter+1,
            }

        case DECREASE:
            return{
                counter:state.counter-1,
            }

        case RESET:
            return{
                counter:0,
            }

        default:
            return state
    }
}
```

Now to create our store and use reducer in it:
```js
import { createStore } from "redux";
import { counterReducer } from "./reducers/counterReducer";
  
export const store=createStore(counterReducer);
```

Now to use it in other components.

Go to main.jsx file:
```jsx
import { StrictMode } from 'react'
import { createRoot } from 'react-dom/client'
import './index.css'
import App from './App.jsx'
import { Provider } from 'react-redux'
import { store } from "./redux/store.js"

createRoot(document.getElementById('root')).render(
  <StrictMode>
    <Provider store={store}>
      <App />
    </Provider>
  </StrictMode>,
)

```
Now the provider lets every component to access our store.

Now let's get to the page that we want to use our logic in it. But first we need to create our actions. So we can use them in store.

Create a folder called actions with a file in it.
Put `export` before the actions you defined in the counter reducer
Now in actions.js file
```jsx
import { INCREASE,DECREASE,RESET } from "../reducers/counterReducer";
  
const increaseCount=()=>( {type:INCREASE} );
  
const decreaseCount=()=>({ type:DECREASE });
  
const resetCount=()=>({ type:RESET });
  
export {increaseCount,decreaseCount,resetCount};
```

Let's get to the file we want to use the logics:
```jsx
import Button from "./components/Button/Button";
import { useState,useReducer } from "react";
import { useDispatch, useSelector } from "react-redux";
import { increaseCount, decreaseCount, resetCount } from "./redux/actions/actions";
  
export default function App(){
  const dispatch=useDispatch();
  const counterValue=useSelector((state)=>state.counter);
  return(
    <div className="w-full h-[100vh] bg-white flex flex-col items-center justify-center gap-8 text-md">
      <p>Counter: {counterValue}</p>
      <div className="flex gap-8">
        <Button onClick={()=>dispatch(increaseCount())}>Increase</Button>
        <Button onClick={()=>dispatch(resetCount())}>Reset</Button>
        <Button onClick={()=>dispatch(decreaseCount())}>Decrease</Button>
      </div>
    </div>
  )
}
```
`useSelector` helps us to access the state we defined in store.
##### Important notes about reducers
Let's do this example. Now we want to add a button that shows or hides the counter and the buttons.
App.jsx
```jsx
import Button from "./components/Button/Button";
import { useState,useReducer } from "react";
import { useDispatch, useSelector } from "react-redux";
import { increaseCount, decreaseCount, resetCount,toggleCount } from "./redux/actions/actions";
  
export default function App(){
  const dispatch=useDispatch();
  const {counter,showCounter}=useSelector((state)=>{
    return{
      counter:state.counter,
      showCounter:state.showCounter
    }
  });
  return(
    <div className="w-full h-screen bg-white flex flex-col items-center justify-center gap-8 text-md">
      {
        showCounter?(
        <>
          <p>Counter: {counter}</p>
          <div className="flex gap-8">
            <Button onClick={()=>dispatch(increaseCount())}>Increase</Button>
            <Button onClick={()=>dispatch(resetCount())}>Reset</Button>
            <Button onClick={()=>dispatch(decreaseCount())}>Decrease</Button>
          </div>
        </>
      ):null}
      <Button variant="secondary" onClick={()=>dispatch(toggleCount())}>Toggle count</Button>
    </div>
  )
}
```
Actions.js
```js
import { INCREASE,DECREASE,RESET, TOGGLE_COUNTER } from "../reducers/counterReducer";
  
const increaseCount=()=>( {type:INCREASE} );
  
const decreaseCount=()=>({ type:DECREASE });
  
const resetCount=()=>({ type:RESET });
  
const toggleCount=()=>({ type:TOGGLE_COUNTER });
  
export {increaseCount,decreaseCount,resetCount,toggleCount};
```

CounterReducer.js
```js
const initialVal={counter:0,showCounter:true};
  
export const INCREASE="increase";
export const DECREASE="decrease";
export const RESET="reset";
export const TOGGLE_COUNTER="toggle_counter";
  
export const counterReducer=(state=initialVal,action)=>{
    switch(action.type){
        case INCREASE:
            return{
                counter:state.counter+1,
            }
  
        case DECREASE:
            return{
                counter:state.counter-1,
            }
  
        case RESET:
            return{
                counter:0,
            }
  
        case TOGGLE_COUNTER:
            return{
                showCounter:!state.showCounter,
            }
  
        default:
            return state
    }
}
```
Now what happens is that the toggle works, but the counter value disappears, or if you refresh and click other buttons, the same thing happens.
It's because when you return a new state, the whole stored datas on the previous state and the new one will not be merged, in other words, it gets overridden. So you need to have a copy of that state. So we can use this structure:
```js
const initialVal={counter:0,showCounter:true};
  
export const INCREASE="increase";
export const DECREASE="decrease";
export const RESET="reset";
export const TOGGLE_COUNTER="toggle_counter";
  
export const counterReducer=(state=initialVal,action)=>{
    switch(action.type){
        case INCREASE:
            return{
	            ...state,
                counter:state.counter+1,
            }
  
        case DECREASE:
            return{
	            ...state,
                counter:state.counter-1,
            }
  
        case RESET:
            return{
	            ...state,
                counter:0,
            }
  
        case TOGGLE_COUNTER:
            return{
	            ...state,
                showCounter:!state.showCounter,
            }
  
        default:
            return state
    }
}
```
*!Note:* You should never mutate your state directly, because states were object, and we know objects and arrays are reference types, which means they are stored in specific memory location. Changing them might result in error.
Directly mutation for example:
```js
case INCREASE:
	state.counter++;
	/*return{
		...state,
		counter:state.counter+1,
	}*/
```
