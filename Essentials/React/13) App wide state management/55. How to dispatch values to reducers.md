We can use the previous matter we discussed about called action's payload.
Which helps us get a value from the ui and do some operations with it.
App
```jsx
import Button from "./components/Button/Button";
import { useState,useReducer } from "react";
import { useDispatch, useSelector } from "react-redux";
import { increaseCount, decreaseCount, resetCount,toggleCount, increaseRandom } from "./redux/actions/actions";
  
export default function App(){
  const dispatch=useDispatch();
  const {counter,showCounter}=useSelector((state)=>{
    return{
      counter:state.counter,
      showCounter:state.showCounter
    }
  });
  return(
    <div className="w-full h-screen bg-white flex flex-col items-center justify-center gap-8 text-md">
      {
        showCounter?(
        <>
          <p>Counter: {counter}</p>
          <div className="flex gap-8">
            <Button onClick={()=>dispatch(increaseCount())}>Increase</Button>
            <Button onClick={()=>dispatch(resetCount())}>Reset</Button>
            <Button onClick={()=>dispatch(decreaseCount())}>Decrease</Button>
            <Button onClick={()=>dispatch(increaseRandom(Math.round(Math.random()*10)))}>Increase by random amount</Button>
          </div>
        </>
      ):null}
      <Button variant="secondary" onClick={()=>dispatch(toggleCount())}>Toggle count</Button>
    </div>
  )
}
```

CounterReducer
```js
const initialVal={counter:0,showCounter:true};
  
export const INCREASE="increase";
export const DECREASE="decrease";
export const RESET="reset";
export const TOGGLE_COUNTER="toggle_counter";
export const INCREASE_RANDOM="increase_random";
  
export const counterReducer=(state=initialVal,action)=>{
    switch(action.type){
        case INCREASE:
            return{
                ...state,
                counter:state.counter+1,
            }
  
        case DECREASE:
            return{
                ...state,
                counter:state.counter-1,
            }
  
        case RESET:
            return{
                ...state,
                counter:0,
            }
  
        case TOGGLE_COUNTER:
            return{
                ...state,
                showCounter:!state.showCounter,
            }
        case INCREASE_RANDOM:
            return{
                ...state,
                counter:state.counter+action.amount_payload,
            }
  
        default:
            return state
    }
}
```

import { INCREASE,DECREASE,RESET, TOGGLE_COUNTER, INCREASE_RANDOM } from "../reducers/counterReducer";

const increaseCount=()=>( {type:INCREASE} );
  
const decreaseCount=()=>({ type:DECREASE });
  
const resetCount=()=>({ type:RESET });
  
const toggleCount=()=>({ type:TOGGLE_COUNTER });
  
const increaseRandom=(randomNum)=>({
    type:INCREASE_RANDOM,
    amount_payload:randomNum,
});
  
export{
    increaseCount,
    decreaseCount,
    resetCount,
    toggleCount,
    increaseRandom,
};
