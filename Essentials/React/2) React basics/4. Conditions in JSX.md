In JSX, unlike the JS you can't use if conditions the traditional way.
```jsx
if(!items.length) return <div>There is no item</div>
```
But this is not a really good way. Cause later on, you may have so many lines of code.

You can use the ternary if as well. But it's not really clean.

Using a function is better
```jsx
const List=()=>{
    // const items=["Tesla","Microsoft","Sony","Apple"];
    const items=[];
    const render_list=()=>{
        if(!items.length) return <div>There is no item</div>;
        return items.map((item)=>{
                    return <li>{item}</li>
                })
    }
    return(
        <>
            <label>Best high tech companies</label>
            <ul>
                {render_list()}
            </ul>
        </>
    );
};
  
export default List;
```

```jsx  
export default function List(){
    const items=["Tesla","Microsoft","Sony","Apple"];
  
    return(
        <>
            <label>The best high tech companies</label>
            <ul>
                {items.length ? (
                    items.map((item,idx)=>(
                        <li key={idx}>{idx}){item}</li>
                    ))
                ):(
                    <p>There is no item(</p>
                )}
            </ul>
        </>
    )
}
```

One of the other solutions is using the `&&` operator:
```jsx
export default function List(){
    const items=["Tesla","Microsoft","Sony","Apple"];
  
    return(
        <>
            <label>The best high tech companies</label>
            <ul>
                {!items.length && <div>There is no item...</div>}
                    {items.map((item,idx)=>(
                        <li key={idx}>
                            {item}
                        </li>
                    ))}
            </ul>
        </>
    )
}
```
