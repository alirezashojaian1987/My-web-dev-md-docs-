When the user enters a wrong address at the address bar, react router will show an error massage. We can style that page when shown.

First you need to create a page and style it for the error massage.
Then you can use the Error element property in your routes page:
```jsx
import { createBrowserRouter } from "react-router-dom";
import Homepage from "../Pages/Homepage/Homepage";
import Shop from "../Pages/Shop/Shop";
import About_us from "../Pages/About_us/About_us";
import Layout from "../components/Layout/index";
import Collection from "../Pages/Collections/Collections";
import Error from "../Pages/Error/Error";
  
export const router=createBrowserRouter([
  {
    path:"/",
    element:<Layout/>,
    errorElement:<Error/>,
    children:[
      {
        index:true,
        element:<Homepage/>,
      },
  
      {
        path:"Shop",
        element:<Shop/>,
      },
  
      {
        path:"About_us",
        element:<About_us/>,
      },
  
      {
        path:"Collections",
        element:<Collection/>,
      }
  
    ]
  }
]);
```

Now a very good example of how to handle and show different errors:
```jsx
import { Link } from "react-router-dom";
import styles from "./Error.module.scss";
import { useRouteError,isRouteErrorResponse } from "react-router-dom";
  
export default function Error(){
    const error=useRouteError();
  
    return(
        <div className={styles.error_page}>
            <div className={styles.content}>
                <h1>Error!</h1>
                {error?.status?<span>{error.status}</span>:null}
                <p>{isRouteErrorResponse(error)?"The page you searched for was not found":"Something went wrong"}</p>
                <Link className={styles.link} to="/">Go back to Homepage</Link>
            </div>
  
            <div className={styles.img}/>
  
        </div>
    )
}
```
