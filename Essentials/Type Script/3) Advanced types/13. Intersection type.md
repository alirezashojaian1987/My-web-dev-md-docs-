Consider this code:
```ts
type Person={name:string; age:number};
type Job={title:string; salary:number};
  
const person:Person={name:"Ali", age:22};
const job:Job={title:"Front-end dev", salary:20000};
  
const createEmployee=(person:Person,job:Job)=>{
    return {...person,...job};
}
```

You want to create an object that has both the `person, job` objects attributes.
But if you pay attention to the code, you ask yourself ok, but now which type should i return as the new object?
Now the `intersection type` comes to our rescue.
```ts
type Person={name:string; age:number};
type Job={title:string; salary:number};
  
const person:Person={name:"Ali", age:22};
const job:Job={title:"Front-end dev", salary:20000};
  
const createEmployee=(person:Person,job:Job):Person & Job=>{
    return {...person,...job};
}
```
or
```ts
type Person={name:string; age:number};
type Job={title:string; salary:number};
type Employee=Person & Job;
 
const person:Person={name:"Ali", age:22};
const job:Job={title:"Front-end dev", salary:20000};
  
const createEmployee=(person:Person,job:Job):Employee=>{
    return {...person,...job};
}

const employee=createEmployee(person,job);
console.log(employee);
```
